// ===== MEASURE: Members =====
Members = DISTINCTCOUNT ( Legislators[Bioguide ID] )

// ===== MEASURE: Senators =====
Senators =
CALCULATE ( [Members], Legislators[Current Chamber] = "Senate" )

// ===== MEASURE: Representatives =====
Representatives =
CALCULATE ( [Members], Legislators[Current Chamber] = "House" )

// ===== MEASURE: Party Share % =====
Party Share % =
DIVIDE ( [Members], CALCULATE ( [Members], ALL ( Legislators[Current Party] ) ) )

// ===== MEASURE: Female Members =====
Female Members =
CALCULATE ( [Members], Legislators[Gender] = "F" )

// ===== MEASURE: Female Share % =====
Female Share % =
DIVIDE ( [Female Members], [Members] )

// ===== MEASURE: Average Age =====
Average Age =
AVERAGEX ( Legislators, DATEDIFF ( Legislators[Birthday], TODAY (), YEAR ) )

// ===== MEASURE: Average Years in Congress =====
Average Years in Congress =
AVERAGEX ( Legislators, DATEDIFF ( Legislators[First Term Start], TODAY (), YEAR ) )

// ===== MEASURE: Average Terms Served =====
Average Terms Served = AVERAGE ( Legislators[Total Terms] )

// ===== MEASURE: Members with Leadership Role =====
Members with Leadership Role =
CALCULATE (
    DISTINCTCOUNT ( LeadershipRoles[Bioguide ID] ),
    ISBLANK ( LeadershipRoles[End] )
)

// ===== MEASURE: Committee Assignments =====
Committee Assignments = COUNTROWS ( MemberCommittees )

// ===== MEASURE: Committee Chairs =====
Committee Chairs =
CALCULATE ( DISTINCTCOUNT ( MemberCommittees[Bioguide ID] ), MemberCommittees[Role Sort] = 1 )

// ===== MEASURE: Ranking Members =====
Ranking Members =
CALCULATE ( DISTINCTCOUNT ( MemberCommittees[Bioguide ID] ), MemberCommittees[Role Sort] = 3 )

// ===== MEASURE: Avg Committees per Member =====
Avg Committees per Member =
DIVIDE (
    CALCULATE ( COUNTROWS ( MemberCommittees ), Committees[Level] = "Committee" ),
    [Members]
)

// ===== MEASURE: Majority Side Members =====
Majority Side Members =
CALCULATE ( DISTINCTCOUNT ( MemberCommittees[Bioguide ID] ), MemberCommittees[Party Side] = "Majority" )

// ===== TABLE: Page =====
Page = GENERATESERIES ( 1, 30, 1 )

// ===== MEASURE: Cards HTML =====
Cards HTML =
VAR PageSize = 24
VAR PageNo   = SELECTEDVALUE ( 'Page'[Page], 1 )
VAR People =
    ADDCOLUMNS (
        VALUES ( MemberCards[Card Sort] ),
        "@Html", CALCULATE ( SELECTEDVALUE ( MemberCards[Card HTML] ) )
    )
VAR UpToThisPage = TOPN ( PageSize * PageNo,       People, MemberCards[Card Sort], ASC )
VAR Earlier      = TOPN ( PageSize * ( PageNo - 1 ), People, MemberCards[Card Sort], ASC )
VAR ThisPage     = EXCEPT ( UpToThisPage, Earlier )
RETURN
    "<style>" & [Card CSS] & "</style><div class='grid'>"
        & CONCATENATEX ( ThisPage, [@Html], "", MemberCards[Card Sort], ASC )
        & "</div>"

// ===== MEASURE: Page Count =====
Page Count =
ROUNDUP ( DIVIDE ( DISTINCTCOUNT ( MemberCards[Card Sort] ), 24 ), 0 )

// ===== MEASURE: Showing Label =====
Showing Label =
VAR N = DISTINCTCOUNT ( MemberCards[Card Sort] )
VAR P = SELECTEDVALUE ( 'Page'[Page], 1 )
RETURN
    IF ( N = 0, "No members match the current filters",
        "Showing " & ( P - 1 ) * 24 + 1 & "–" & MIN ( P * 24, N ) & " of " & N & " members" )
